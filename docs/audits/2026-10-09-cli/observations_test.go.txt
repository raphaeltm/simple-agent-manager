func TestAuditChatPaginationFlagsIgnored(t *testing.T) {
	for _, jsonMode := range []bool{false, true} {
		args := []string{"chat", "session_fixture", "--project=01KHRJGANBBWGDY1NZ0KVF0D4J", "--limit=1", "--before=123"}
		if jsonMode {
			args = append(args, "--json")
		}
		calls := 0
		doer := roundTripFunc(func(r *http.Request) (*http.Response, error) {
			calls++
			if r.Method != "GET" || r.URL.RawQuery != "" {
				t.Fatal("unexpected paging request")
			}
			return jsonResponse(`{"session":{"id":"session_fixture"},"messages":[{"id":"message_fixture","role":"assistant","content":"Synthetic page","createdAt":123,"sequence":7,"origin":"system","toolMetadata":{"content":[]}}],"hasMore":true}`, 200), nil
		})
		runtime, out, errout := testRuntime(t, args, doer, nil)
		if Run(context.Background(), runtime) != 0 || calls != 1 {
			t.Fatalf("unexpected result: %s", errout.String())
		}
		if jsonMode {
			var got map[string]any
			if json.Unmarshal(out.Bytes(), &got) != nil {
				t.Fatal("invalid JSON")
			}
			if got["hasMore"] != true {
				t.Fatal("expected hasMore")
			}
			m := got["messages"].([]any)[0].(map[string]any)
			for _, key := range []string{"sequence", "origin", "toolMetadata"} {
				if _, ok := m[key]; ok {
					t.Fatalf("unexpected retained %s", key)
				}
			}
		} else if strings.Contains(out.String()+errout.String(), "hasMore") || strings.Contains(out.String()+errout.String(), "more messages") {
			t.Fatal("unexpected truncation warning")
		}
	}
}

func TestAuditJSONFieldLoss(t *testing.T) {
	cases := []struct {
		name   string
		args   []string
		body   string
		absent []string
	}{
		{"projects", []string{"projects"}, `{"projects":[],"nextCursor":"next_fixture"}`, []string{"nextCursor"}},
		{"sessions", []string{"chat"}, `{"sessions":[],"total":21}`, []string{"total"}},
		{"ideas", []string{"ideas"}, `{"tasks":[{"id":"idea_fixture","description":"Synthetic description"}],"nextCursor":"next_fixture"}`, []string{"nextCursor", "description"}},
		{"profile", []string{"profiles"}, `{"items":[{"id":"profile_fixture","name":"Fixture","model":"fixture-model","effort":"high","runtime":"vm","projectId":null,"permissionMode":"default"}]}`, []string{"model", "effort", "runtime", "projectId", "permissionMode"}},
		{"activity", []string{"activity"}, `{"events":[],"hasMore":true}`, []string{"hasMore"}},
		{"notification", []string{"notifications"}, `{"notifications":[{"id":"notification_fixture","body":"Synthetic body","projectId":"project_fixture","actionUrl":"/fixture"}]}`, []string{"body", "projectId", "actionUrl"}},
		{"project", []string{"project"}, `{"id":"project_fixture","summary":{"activeWorkspaceCount":3},"defaultAgentType":"openai-codex"}`, []string{"summary", "defaultAgentType"}},
		{"task", []string{"task", "status", "task_fixture"}, `{"id":"task_fixture","description":"Synthetic description","chatSessionId":"session_fixture","blocked":true,"dependencies":[]}`, []string{"description", "chatSessionId", "blocked", "dependencies"}},
	}
	for _, c := range cases {
		t.Run(c.name, func(t *testing.T) {
			args := append(c.args, "--project=01KHRJGANBBWGDY1NZ0KVF0D4J", "--json")
			doer, _ := captureJSONRequest(t, c.body, http.StatusOK)
			runtime, out, errout := testRuntime(t, args, doer, nil)
			if Run(context.Background(), runtime) != 0 {
				t.Fatalf("failed: %s", errout.String())
			}
			var got map[string]any
			if json.Unmarshal(out.Bytes(), &got) != nil {
				t.Fatal("invalid JSON")
			}
			rendered := out.String()
			for _, key := range c.absent {
				if strings.Contains(rendered, `"`+key+`"`) {
					t.Fatalf("unexpected retained %s", key)
				}
			}
		})
	}
}
